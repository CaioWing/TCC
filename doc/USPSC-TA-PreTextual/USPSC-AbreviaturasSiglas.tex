% USPSC-AbreviaturasSiglas.tex
\begin{siglas}
    \setlength{\itemsep}{2pt}
    \item[AP]    Precisão média (\textit{Average Precision})
    \item[AVI] Formato de arquivo de áudio e vídeo (\textit{Audio Video Interleave})
    \item[COCO]  \textit{Common Objects in Context} (base de imagens)
    \item[COPPE] Instituto Alberto Luiz Coimbra de Pós-Graduação e Pesquisa de Engenharia da UFRJ
    \item[CPU]   Unidade central de processamento
    \item[CSV] Formato de dados tabulares (\textit{Comma-Separated Values})
    \item[CUDA]  Plataforma da NVIDIA para computação paralela em GPU
    \item[CVAT] Ferramenta de anotação de imagens e vídeos (\textit{Computer Vision Annotation Tool})
    \item[DVFS]  Ajuste dinâmico da tensão e da frequência de operação
    \item[FN]    Falso negativo
    \item[FP]    Falso positivo
    \item[FP16, FP32] Representação em ponto flutuante com 16 ou 32 bits
    \item[GPS]   Sistema de posicionamento global (\textit{Global Positioning System})
    \item[GPU]   Unidade de processamento gráfico
    \item[GSD]   Distância no solo correspondente a um pixel (\textit{Ground Sample Distance})
    \item[IBGE] Instituto Brasileiro de Geografia e Estatística
    \item[INT8]  Representação inteira de 8 bits
    \item[IoU]   Razão entre as áreas de interseção e união de duas regiões (\textit{Intersection over Union})
    \item[L4T] Distribuição Linux para a plataforma Tegra (\textit{Linux for Tegra})
    \item[mAP]   Média das precisões médias das classes (\textit{mean Average Precision})
    \item[MBG-V2] Segunda versão da base de dados \textit{Mosquito Breeding Grounds}
    \item[MJPG] Codificação de vídeo \textit{Motion JPEG}
    \item[NMS]   Supressão de caixas sobrepostas (\textit{Non-Maximum Suppression})
    \item[ONNX]  Formato para intercâmbio de modelos de redes neurais (\textit{Open Neural Network Exchange})
    \item[ORT]   ONNX Runtime, ambiente de execução de modelos ONNX
    \item[PNG] Formato de imagem (\textit{Portable Network Graphics})
    \item[RANSAC] Método de ajuste por consenso entre amostras aleatórias (\textit{Random Sample Consensus})
    \item[SIFT]  Método para identificar e descrever pontos de uma imagem (\textit{Scale-Invariant Feature Transform})
    \item[UFRJ]  Universidade Federal do Rio de Janeiro
    \item[VANT]  Veículo aéreo não tripulado
    \item[VP]    Verdadeiro positivo
    \item[WGS-84] Sistema geodésico mundial de 1984 (\textit{World Geodetic System 1984})
    \item[XML] Linguagem de marcação (\textit{Extensible Markup Language})
    \item[YOLO]  Família de detectores de objetos (\textit{You Only Look Once})
\end{siglas}
