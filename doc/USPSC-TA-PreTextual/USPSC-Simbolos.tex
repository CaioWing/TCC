\begin{simbolos}
  \item[$A$]                Área do disco do rotor
  \item[$A_t$]              Área de solo coberta por um recorte de imagem
  \item[$b$]                Borrão de movimento, em pixels
  \item[$C_{d0}$]           Coeficiente de arrasto de perfil médio das pás
  \item[$D$]                Arrasto da aeronave
  \item[$e$]                Energia por unidade de área inspecionada
  \item[$E_b$]              Energia útil da bateria
  \item[$E_{\mathrm{inf}}$] Energia adicional por passada da rede neural
  \item[$f_c$]              Distância focal da câmera
  \item[$f_e$]              Área equivalente de placa plana
  \item[$f_q$]              Taxa de quadros do vídeo
  \item[$FM$]               Figura de mérito do rotor
  \item[$g$]                Aceleração da gravidade
  \item[$GSD$]              Resolução espacial no solo
  \item[$h$]                Altura de voo sobre o terreno
  \item[$m$]                Massa da aeronave
  \item[$n_t$]              Número de recortes por quadro
  \item[$n_v$]              Número de quadros em que um ponto do solo é visto
  \item[$P$]                Potência mecânica requerida
  \item[$P_c$]              Potência do computador de bordo
  \item[$P_{el}$]           Potência elétrica retirada da bateria
  \item[$P_h$]              Potência de pairado
  \item[$r$]                Taxa de passadas da rede neural
  \item[$\mathcal{R}_f$]    Revocação de focos únicos
  \item[$T$]                Tração do rotor
  \item[$t_e$]              Tempo de exposição
  \item[$v_h$]              Velocidade induzida em pairado
  \item[$v_i$]              Velocidade induzida
  \item[$V$]                Velocidade de voo
  \item[$w$]                Largura da faixa imageada
  \item[$\alpha_d$]         Ângulo de ataque do disco do rotor
  \item[$\eta_m$]           Eficiência combinada de motores e controladores
  \item[$\theta$]           Ângulo de arfagem
  \item[$\kappa$]           Fator de correção da potência induzida
  \item[$\mu$]              Razão de avanço
  \item[$\rho$]             Densidade do ar
  \item[$\sigma$]           Solidez do rotor
  \item[$\psi$]             Rumo da câmera
  \item[$\Omega R$]         Velocidade de ponta da pá
\end{simbolos}
